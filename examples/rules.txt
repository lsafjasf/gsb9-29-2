# 示例访问策略（动作 协议 源网段 目的网段 端口）
allow tcp  10.0.0.0/8    any           80        # 规则1: 内网 Web
allow tcp  10.0.0.0/8    any           443       # 规则2: 内网 HTTPS
deny  tcp  any           192.168.1.10  22        # 规则3: 禁止 SSH 到核心服务器
allow tcp  10.1.0.0/16   192.168.1.10  22        # 规则4: 想放行运维网段 SSH，但被规则3遮蔽
allow tcp  10.0.0.0/8    any           80        # 规则5: 与规则1 完全重复
allow udp  10.0.0.0/8    any           53        # 规则6: 内网 DNS
allow tcp  10.0.0.0/16   any           1-65535   # 规则7: 与规则1/2 部分重叠但同动作，无问题
deny  any  172.16.0.0/12 any           any       # 规则8: 屏蔽 172.16 私网段全部流量
allow tcp  172.16.5.0/24 any           25        # 规则9: 想放行邮件，被规则8 遮蔽
allow tcp  10.0.0.0/16   any           8080      # 规则10
allow tcp  10.1.0.0/16   any           8080      # 规则11
allow tcp  10.0.0.0/15   any           8080      # 规则12: 被规则10+11 联合覆盖
